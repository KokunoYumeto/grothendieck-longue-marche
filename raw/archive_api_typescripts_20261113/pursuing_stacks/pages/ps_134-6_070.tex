\[ \hand{$\mathcal{A}$}_{ab} \hand{$\xrightarrow{\approx}$} \underline{Hom}^{\hand{!}}(\hand{$\mathcal{A}$}_{ab}{}^{o}, \text{(Sets)}) \quad ( \hand{$\xleftarrow{\approx}$} \underline{Hom}^{\hand{!}}(\hand{$\mathcal{A}$}_{ab}{}^{o}, \text{(Ab)}) ) \quad , \]
