% These wrappers deliberately do not box their contents: paragraphs, nested
% editorial macros, and figures may occur inside deleted or inserted content.
% Color is a QA annotation, not a proposed rendering of manuscript ink.

\NewDocumentCommand{\unsure}{+m}{{\color{QADoubt}#1}}

  \ifmmode{\color{QAMargin}#1}%
  \else\par\smallskip\begingroup\color{QAMargin}%
    \noindent\textsf{[Margin]}\quad #1\par\endgroup\smallskip\fi}

% ulem is absent in the present TeX installation. If later available, use it.
% The fallback preserves visible single/double underlining but cannot break an
% underlined phrase across lines; an overfull-box warning remains useful in QA.
% Define argument-taking helpers outside IfFileExists: its implementation may
% store the selected branch in a temporary macro before executing that branch.
\NewDocumentCommand{\QAUnderlineFallback}{+m}{%
  \ifmmode\underline{#1}\else\underline{\strut #1}\fi}
\NewDocumentCommand{\QADoubleUnderlineFallback}{+m}{%
  \ifmmode\underline{\underline{#1}}%
  \else\underline{\underline{\strut #1}}\fi}
\IfFileExists{ulem.sty}{%
  \usepackage[normalem]{ulem}%
}{%
  \let\uline\QAUnderlineFallback
  \let\uuline\QADoubleUnderlineFallback
}
