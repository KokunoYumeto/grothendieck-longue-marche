\unsure{$k$-Algèbre} différentielle $C^*$ sur \unsure{Topos} $\unsure{\mathcal T|T}$.

Anneau de base $k$ (commutatif)
résolution $C^*$ de $k$

\margin{\deleted{\unsure{avec résolution} \illegible{2} \unsure{d'augmentation}}}

\diagram{\begin{tikzcd}
\unsure{0|-} \arrow[r] & C^0 \arrow[r] & C^1 \arrow[r] & C^2 \arrow[r] & \unsure{-|\cdots}\\
& k \arrow[u]
\end{tikzcd}}

(la section unité de $k$ correspond à une section $1$ de $C^0$)

\deleted{\unsure{bilin.}}

Accouplements (notés $x,y\longmapsto x\circ y$)
\[
\left\{\begin{array}{lll}
Z^i\times C^j\longrightarrow C^{i+j}&(a_{ij})\\
C^i\times Z^j\longrightarrow C^{i+j}&(b_{ij})&i,j\geq0
\end{array}\right.
\]
satisfaisant aux conditions

$1^\circ)$ $a_{ij}$ et $b_{ij}$ coïncident sur $Z^i\times Z^j$ et
y prennent valeurs dans $Z^{i+j}$. (D'où la notation
$x\circ y$ non ambiguë, elle a un sens si $dx$ \uline{ou} $dy$ est \unsure{nulle}.)
\deleted{\unsure{donc}}

$2^\circ)$ Associativité $(x\circ y)\circ z=x\circ(y\circ z)$ \unsure{si deux}
des trois sections locales $x,y,z$ (de $C^i,C^j,C^h$)
sont des cocycles; $1\circ x=x\circ1=x$.

\[
3^\circ)\quad\left\{\begin{array}{l}
x\circ y=y\circ x\;(-1)^{\deg y\deg x}\\
x\circ x=0\quad\text{si $x$ section locale de $Z^i$, $i$ \uline{impair}}
\end{array}\right.
\]

\deleted{si $\unsure{x^i|u^i}$ \unsure{section locale de} $C^i$, $\unsure{z^j|x^j}$ \unsure{section locale de} $Z^j$}

$4^\circ)$ Pour tout $i,j\geq1$, existe un $h_{ij}$ \unsure{bilin.}
\deleted{\diagram{\begin{tikzcd}
C^{i-1}\times Z^j\arrow[r,"h_{ij}"] & C^{i+j-1}
\end{tikzcd}}}
\inserted{$\unsure{Z^i\times Z^j|Z^i}$}
\unsure{tel que (ayant pour assertion le)}
\diagram{\begin{tikzcd}
Z^i\times Z^j\arrow[r] & Z^{i+j}\\
C^{i-1}\times Z^j\arrow[u,"d\times\mathrm{id}"]\arrow[r,"h_{ij}"'] & C^{i+j-1}\arrow[u,"d"']
\end{tikzcd}}
\margin{$\unsure{h_{ij}(x^i\circ z^j)=\mathrm{id}\circ d|h(x^i\circ z^j)=d}$}

\deleted{(donc \unsure{si} $i,j\unsure{\geq|>}0$, \unsure{on a}
\[
\unsure{h_{ij}(dx^{i-1})\times z^j=d(h_{ij}(x^{i-1},z^j))}
\]
\unsure{si} $x^{i-1}$ \unsure{section locale} \illegible{3})}

\deleted{(\uline{NB} l'indétermination des $h_{ij}$ \unsure{est dans}
\[
\operatorname{Hom}_k(C^{i-1}\otimes_k Z^j,Z^{i+j-1});
\]
\unsure{celle-ci est} \illegible{2} \unsure{voulue}; on peut \unsure{exiger pour les}
\illegible{2} \unsure{donnés}, et pour des \illegible{3}.)}
