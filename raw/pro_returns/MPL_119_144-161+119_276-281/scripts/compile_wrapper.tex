% \mathpalette preserves display/text/script size for a mathematical strikeout.
% The text branch also accepts inline mathematics in dollar delimiters.
\newcommand{\ms@mathstrike}[2]{%
  \hbox{\sout{\ensuremath{#1#2}}}}
\DeclareRobustCommand{\deleted}[1]{%
  \ifmmode{\color{MSDeletion}\mathpalette\ms@mathstrike{#1}}%
  \else{\color{MSDeletion}\sout{#1}}\fi}

\DeclareRobustCommand{\inserted}[1]{%
  \ifmmode
    {\color{MSInsertion}\mathopen{\langle}{}^{+}#1\mathclose{\rangle}}%
  \else
    {\color{MSInsertion}\ensuremath{\langle^{+}}#1\ensuremath{\rangle}}%
  \fi}
\DeclareRobustCommand{\unsure}[1]{%
  \ifmmode
    {\color{MSDoubt}\mathopen{\langle}?\,#1\mathclose{\rangle}}%
  \else
    {\color{MSDoubt}\ensuremath{\langle}\textbf{?}\,#1\ensuremath{\rangle}}%
  \fi}
\DeclareRobustCommand{\illegible}[1]{%

  \ifmmode{\color{gray}\text{[margin: }#1\text{]}}%
  \else{\color{gray}\textsf{[margin: }#1\textsf{]}}\fi}
