\[

\[
\operatorname{Hom}(C,C')\longrightarrow
\operatorname{Hom}(\underbrace{\operatorname{Ob}C}_{C_0},\underbrace{\operatorname{Ob}C'}_{C'_0}).

\[ f_0:C_0\longrightarrow C'_0 \]
