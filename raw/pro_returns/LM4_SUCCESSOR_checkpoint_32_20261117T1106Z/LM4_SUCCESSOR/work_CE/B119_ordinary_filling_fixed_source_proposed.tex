\subsection*{The ordinary one-neck presentation and its retained quotient}

\statement{B.119}{Proved}{The actual one-neck ordinary group has a complete thirteen-generator presentation, and its eight-family filling has exactly the actual free-two fixed source in both full categories}
\label{stmt:B119}
Let $\mathcal G=G_{\rm ann}$ be the actual orbifold group of B.114,
over the two complete annuli joined through the first regular neck.
Use its actual common disk marking
$K=F(a,B,d,e,f)$, actual selected gap cones
$c=c_-$ and $\tau=c_+$, and actual far-section loops
$m=k_-$, $p=k_+$. Let $\mathcal G^0$ be the ordinary index-two
subgroup: $c,\tau$ are odd and $K,m,p$ are even. Put
\[
 \widetilde u=cuc\quad(u=a,B,d,e,f),\qquad
 g=\widetilde f,\qquad x=\tau c.
\]
Then $\mathcal G^0$ has the following complete marked presentation:
thirteen generators
\begin{equation}\label{eq:B119-thirteen}
 a,B,d,e,f,\ \widetilde a,\widetilde B,\widetilde d,
 \widetilde e,g,\ m,p,x,
\end{equation}
and the sixty-four regional relators specified by the explicit
two-sheet rewriting scheme in the proof. The two cross commutators
are eliminated by exactly
\begin{equation}\label{eq:B119-cross-elimination}
 cmc=x^{-1}mx,\qquad cpc=p.
\end{equation}
No regional relator or signed base-loop consequence is discarded.

Let $\mathcal N$ be the ordinary normal closure of the eight actual
deleted-family meridians
\[
 a,B,d,e,\widetilde a,\widetilde B,\widetilde d,\widetilde e,
\]
and let $Q^0=\mathcal G^0/\mathcal N$ be the actual filling of
both families. Then there is an exact marked isomorphism
\begin{equation}\label{eq:B119-filled-semidirect}
 Q^0=F(f,g,x)\rtimes F(m,p),
\end{equation}
where conjugation by $m,p$ acts on the free fiber by
\begin{equation}\label{eq:B119-base-actions}
 \begin{array}{c|ccc}
       &f&g&x\\ \hline
 \alpha_m&f^{-1}gf&f&xg^{-1}f\\
 \alpha_p&fxgx^{-1}f^{-1}&gx^{-1}fxg^{-1}&fxg^{-1}.
 \end{array}
\end{equation}
The six actual positive source generators from B.114 map, in their
actual order, to
\begin{equation}\label{eq:B119-source-images}
 1,\ 1,\ 1,\ 1,\ v=f^{-1}p,\ m.
\end{equation}
Write $S=\langle m,v\rangle\leq Q^0$. Then $S\cong F_2$ is an
actual source splitting of the geometric base projection, and
conjugation $\delta=\operatorname{Inn}(\tau)$ satisfies
\begin{equation}\label{eq:B119-fixed-equalities}
 (Q^0)^\delta=S,
 \qquad (\widehat{Q^0})^\delta=\widehat S.
\end{equation}
All completions are full profinite completions, and $S$ has its full
induced topology. Equivalently, for the filled orbifold group
$Q=Q^0\rtimes\langle\tau\rangle$,
\begin{equation}\label{eq:B119-orbifold-centralizers}
 C_Q(\tau)=S\times\langle\tau\rangle,
 \qquad
 C_{\widehat Q}(\tau)=\widehat S\times\langle\tau\rangle.
\end{equation}
These are assertions about the actual eight-family filling. They do
not give a splitting back into $\mathcal G^0$, a fixed-element lift
through $\mathcal N$, or a centralizer formula for the family-unfilled
one-neck group.

\begin{proof}
\emph{B.119.1. All regional and cross relators in one actual gauge.}
We give the full generating scheme so that the thirteen-generator
presentation is determined without omitting any long word. The physical
neck basis is
\begin{equation}\label{eq:B119-O-marking}
 \begin{gathered}
 O_1=aBa^{-1},\quad O_2=a,\quad
 O_3=fde d^{-1}f^{-1},\quad O_4=d,\quad O_5=d^{-1}fd,\\
 P_i=O_1\cdots O_i,\qquad P_0=1.
 \end{gathered}
\end{equation}
The actual based mirror from B.114 and its certified whole-disk
comparison has
\begin{equation}\label{eq:B119-eta}
 \eta(O_i)=P_{i-1}O_i^{-1}P_{i-1}^{-1},\qquad \eta^2=1.
\end{equation}
The inverse $O$-marking is
\[
 a=O_2,\quad B=O_2^{-1}O_1O_2,\quad d=O_4,\quad
 f=O_4O_5O_4^{-1},\quad e=O_5^{-1}O_4^{-1}O_3O_4O_5.
\]
Thus \eqref{eq:B119-O-marking}--\eqref{eq:B119-eta} specify every
outer word literally in the same free $K$, with no separate
peripheral conjugation convention.

For each sign define
\begin{equation}\label{eq:B119-regional-substitutions}
 \begin{gathered}
 (a_-,B_-,d_-,e_-,f_-)=(a,B,d,e,f),\\
 (a_+,B_+,d_+,e_+,f_+)
       =(\eta(a),\eta(B),\eta(d),\eta(e),\eta(f)),\\
 k_-=m,\quad k_+=p,\qquad
 C_-=a^{-1}ca,\quad C_+=a_+^{-1}\tau a_+,\\
 b_\pm=C_\pm B_\pm C_\pm,\qquad h_\pm=f_\pm k_\pm.
 \end{gathered}
\end{equation}
The substitutions for $C_\pm$ are the inverse of the actual selected
upper-gap cone comparisons in B.114. They change the two cone words
while keeping the entire common $K$ fixed.

For a regional tuple $(a,b,d,e,C,h,k)$ let $\mathscr R$ be the
sixteen words expressing
\begin{equation}\label{eq:B119-sixteen-scheme}
 \begin{gathered}
 [h,C]=1,\qquad hkh^{-1}=CkC,\qquad
 [k,U]=1\quad(U=a,b,d,e),\\
 [U,CUC]=1\quad(U=a,b,d,e),\\
 [a,b]=[CaC,d]=[CaC,e]=[d,CeC]=1,\\
 [b,hdh^{-1}]=[b,heh^{-1}]=1.
 \end{gathered}
\end{equation}
An equality $u=v$ in this list denotes the relator $uv^{-1}$,
and $[u,v]=uvu^{-1}v^{-1}$. Substitute
\eqref{eq:B119-regional-substitutions} into this list to obtain
$\mathscr R_-$ and $\mathscr R_+$. The complete actual presentation
from B.112 and B.114 is
\begin{equation}\label{eq:B119-nine-thirty-six}
 \mathcal G=
 \frac{F(a,B,d,e,f,m,p)*C_2(c)*C_2(\tau)}
 {\langle\!\langle\mathscr R_-,\mathscr R_+,
                    [\tau,m],[c,p]\rangle\!\rangle}.
\end{equation}
It has thirty-six relators when the two cone squares are counted in
the free-group presentation. This is the actual one-neck quotient,
not the proper amalgam before its cross relations.

\emph{B.119.2. An explicit whole-cover rewriting.}
Before the regional and cross relations are imposed, the ordinary
kernel of the free product in \eqref{eq:B119-nine-thirty-six} is
free on fifteen generators
\[
 u,\widetilde u\ (u=a,B,d,e,f),\quad
 m,p,\widetilde m=cmc,\widetilde p=cpc,\quad x=\tau c.
\]
Here is a complete rewriting rule using the transversal $\{1,c\}$.
Keep a state $s\in\{0,1\}$, representing a final $c^s$. An even
letter $u^{\pm1}$ contributes $u^{\pm1}$ in state zero and
$\widetilde u^{\pm1}$ in state one, without changing state.
The letter $c$ changes state and contributes nothing. The letter
$\tau$ contributes $x$ in state zero and $x^{-1}$ in state one,
and changes state. Inverse cone letters are the same cone letters.
This is simply the identity $\tau=xc$, together with
$cxc=x^{-1}$. It gives inverse free-product maps, so it proves
the stated free basis as well as an algorithm for all words.

For any even relator $r$, let $\operatorname{rw}_s(r)$ be this
rewrite starting in state $s$. Its terminal state is again $s$;
the two outputs represent $r$ and $crc$, respectively. The cross
relators give
\[
 x\widetilde m x^{-1}m^{-1},\quad
 x^{-1}mx\widetilde m^{-1},\qquad
 \widetilde p p^{-1},\quad p\widetilde p^{-1}.
\]
They eliminate $\widetilde m=x^{-1}mx$ and $\widetilde p=p$;
the companion words then vanish. Denote this substitution by $E$.
The promised complete list of sixty-four ordinary relators is
\begin{equation}\label{eq:B119-sixty-four}
 \left\{E\operatorname{rw}_s(r):
       r\in\mathscr R_-\cup\mathscr R_+,\quad s=0,1\right\}.
\end{equation}
All its inputs and operations have been specified explicitly above.
This is the full Reidemeister--Schreier presentation followed by
two generator eliminations, and proves
\eqref{eq:B119-thirteen}--\eqref{eq:B119-cross-elimination}.

Conjugation by $c$ induces the exact free-group involution on these
thirteen generators
\begin{equation}\label{eq:B119-minus-cone-action}
 u\longleftrightarrow\widetilde u,\qquad x\longmapsto x^{-1},
 \qquad m\longmapsto x^{-1}mx,\qquad p\longmapsto p.
\end{equation}
It preserves \eqref{eq:B119-sixty-four} as a normal subgroup.
Conjugation by $\tau=xc$ is its composition with
$\operatorname{Inn}(x)$. Both squares are the identity already
on the free thirteen-generator group. In particular these are
the actual two cone actions, not actions inferred only on a quotient
of the abelianization.

\emph{B.119.3. The geometric family filling and all surviving words.}
The first four meridians in the chosen disk are the actual four
deleted-family punctures. Their $c$-partners are the four punctures
in the other family, with the common connectors retained. Filling
those divisors kills their ordinary normal closure $\mathcal N$.
The subgroup $\mathcal N$ is invariant under
\eqref{eq:B119-minus-cone-action}, so it is also the intersection
with $\mathcal G^0$ of the corresponding orbifold meridian normal
closure. This is the usual based filling comparison by meridians;
it does not require an injection of a newly filled fiber.

Set $a,B,d,e,\widetilde a,\widetilde B,\widetilde d,
\widetilde e$ equal to $1$ in \eqref{eq:B119-sixty-four}.
Equations \eqref{eq:B119-O-marking}--\eqref{eq:B119-eta} then give
$f_+=f^{-1}$. Of all sixty-four relators, exactly the following
eight remain nonempty as freely reduced words:
\begin{equation}\label{eq:B119-eight-words}
 \begin{aligned}
 M_1&=fmx^{-1}m^{-1}xg^{-1},&
 M_2&=gx^{-1}mxm^{-1}f^{-1},\\
 M_3&=fmf^{-1}x^{-1}m^{-1}x,&
 M_4&=gx^{-1}mxg^{-1}m^{-1},\\
 P_1&=f^{-1}pxp^{-1}gx^{-1},&
 P_2&=g^{-1}px^{-1}p^{-1}fx,\\
 P_3&=f^{-1}pfxp^{-1}x^{-1},&
 P_4&=g^{-1}pgx^{-1}p^{-1}x.
 \end{aligned}
\end{equation}
The identities $M_2=M_1^{-1}$ and
$P_2=x^{-1}P_1^{-1}x$ are literal free-word identities.
No nonempty node relation is suppressed in passing to this list:
all ten regional node relations on each side vanish under the
specified family filling, as do the four family/base commutators.
The surviving words come from the two generic cone/base relations
on each side and their two ordinary rewrites.

\emph{B.119.4. Both directions of the semidirect presentation.}
The three equations $M_1=M_3=M_4=1$ are equivalent to
\begin{equation}\label{eq:B119-m-three}
 mxm^{-1}=xg^{-1}f,\qquad
 m(xf)m^{-1}=xf,\qquad m(gx^{-1})m^{-1}=gx^{-1}.
\end{equation}
For example $M_1=1$ solves for the conjugate of $x^{-1}$,
and $M_3,M_4$ say that the two displayed products are fixed.
Solving these equations for the conjugates of $f,g,x$ gives
the first row of \eqref{eq:B119-base-actions}; conversely that row
implies all three equations and both $M_1,M_2$.
Likewise $P_1=P_3=P_4=1$ is equivalent to
\begin{equation}\label{eq:B119-p-three}
 pxp^{-1}=fxg^{-1},\qquad
 p(fx)p^{-1}=fx,\qquad p(gx^{-1})p^{-1}=gx^{-1},
\end{equation}
which gives exactly the second row of
\eqref{eq:B119-base-actions} and conversely all four $P_i$.

The rows define actual automorphisms of the free group $F(f,g,x)$.
Their inverse substitutions are
\begin{equation}\label{eq:B119-base-inverses}
 \begin{array}{c|ccc}
       &f&g&x\\ \hline
 \alpha_m^{-1}&g&gfg^{-1}&xfg^{-1}\\
 \alpha_p^{-1}&xgx^{-1}&x^{-1}fx&xg^{-1}x^{-1}fx.
 \end{array}
\end{equation}
Both inverse compositions reduce freely to the original generators.
One can also use the free basis $(y=fx,z=gx^{-1},x)$ for the
second row: it fixes $y,z$ and sends $x$ to $yx^{-1}z^{-1}$,
with inverse $x\mapsto z^{-1}x^{-1}y$.
Thus \eqref{eq:B119-eight-words} is equivalent in both directions
to the six mapping relations for a free base on $m,p$ acting by
these automorphisms. This proves the exact isomorphism
\eqref{eq:B119-filled-semidirect}, including injectivity and freeness
of the fiber and the indicated far-section splitting. It is more
than a verification of a map onto a semidirect product.

\emph{B.119.5. The actual source and its fixed splitting.}
We specify the raw source comparison before filling. In the outer tuple
\eqref{eq:B119-regional-substitutions}, put
\begin{equation}\label{eq:B119-raw-source-factors}
 \begin{gathered}
 E_+=f_+d_+e_+d_+^{-1}f_+^{-1},\\
 W_+=C_+b_+C_+b_+C_+E_+C_+b_+^{-1}C_+b_+^{-1},\\
 \Delta_U=UC_+UC_+,\qquad
 \Gamma_U=C_+b_+[U,C_+UC_+]b_+^{-1}C_+\Delta_U.
 \end{gathered}
\end{equation}
B.112's literal positive source calculation, followed by the whole
selected-gap conjugation, gives the ordered six words
\begin{equation}\label{eq:B119-source-before-filling}
 \begin{gathered}
 a_+(C_+b_+C_+b_+)a_+^{-1},\qquad a_+W_+^2a_+^{-1},\\
 a_+\Gamma_{d_+}a_+^{-1},\qquad
 a_+\Gamma_{a_+}a_+^{-1},\qquad
 a_+h_+a_+^{-1},\qquad m.
 \end{gathered}
\end{equation}
These are the actual common-based source words; the conjugation of
each entire word is retained. All six agree with the original stored
positive lasso words after free cancellation and cone squares alone.
The triangular source images in B.112 also represent these same
elements, but their first four reductions use the established node
relations; no free-word equality with that reduced table is assumed
here. Under the eight-family filling all four outsider
letters in the outer tuple vanish, $W_+$ maps to $C_+$, each
displayed $\Gamma_U$ maps to $1$, $f_+=f^{-1}$, and
$h_+=f^{-1}p$. Hence \eqref{eq:B119-source-before-filling} gives
exactly \eqref{eq:B119-source-images}. The same images have been
checked directly on all six original stored positive words.

The actual plus-cone action, obtained from
\eqref{eq:B119-minus-cone-action} and $\tau=xc$, is
\begin{equation}\label{eq:B119-plus-cone-action}
 \begin{gathered}
 \delta(f)=xgx^{-1},\qquad \delta(g)=xfx^{-1},\qquad
 \delta(x)=x^{-1},\\
 \delta(m)=m,\qquad \delta(p)=xpx^{-1}.
 \end{gathered}
\end{equation}
In particular $m$ is fixed. For $v=f^{-1}p$,
\[
 \delta(v)=xg^{-1}px^{-1}=f^{-1}p,
\]
where the last equality is the actual relation
$\alpha_p(x^{-1})=gx^{-1}f^{-1}$ from
\eqref{eq:B119-base-actions}. Thus $\delta$ fixes both actual
source lifts. They project to the two free base generators $m,p$.
The homomorphism from the free base sending these generators to
$m,v$ has the projection as a left inverse; it is therefore an
injective splitting. Consequently
\begin{equation}\label{eq:B119-source-splitting}
 Q^0=F(f,g,x)\rtimes S,\qquad S=F(m,v),\qquad
 \delta|_S=1.
\end{equation}
The action of $v$ on the fiber is
$\operatorname{Inn}(f^{-1})\alpha_p$, not $\alpha_p$ itself.
The four contact generators disappear under this filling; no
injectivity of the original six-generator source into
$\mathcal G^0$ or into this quotient is being inferred.

\emph{B.119.6. The retained fiber and the exact discrete fixed group.}
The retained orbifold fiber, with these same based meridians, is
\begin{equation}\label{eq:B119-orbifold-fiber}
 E=C_2(c)*C_2(\tau)*\mathbb Z(f).
\end{equation}
Its ordinary index-two kernel is freely generated by
$f,cfc,\tau c$, exactly $f,g,x$. The inverse substitution
$\tau=xc$ and the action
$cfc=g$, $cgc=f$, $cxc=x^{-1}$ identify this free product with
the fiber extension $F(f,g,x)\rtimes\langle c\rangle$ inside
the filled group. The preceding presentation equivalence guarantees
that no extra fiber relation has been assumed away.

The centralizer in a free product of a nontrivial element of a
specified order-two factor is that factor: reduction of a conjugate
by a word outside it retains a syllable outside the factor. Hence
$C_E(\tau)=\langle\tau\rangle$. Its intersection with the ordinary
fiber is trivial. Thus the restriction of $\delta$ to $F(f,g,x)$
has trivial fixed subgroup. By the unique decomposition
$ns$ in \eqref{eq:B119-source-splitting},
$\delta(ns)=\delta(n)s$ is equal to $ns$ precisely when $n=1$.
This proves $(Q^0)^\delta=S$ and the discrete equality in
\eqref{eq:B119-orbifold-centralizers}.

\emph{B.119.7. Cofinal finite quotients and full completion.}
Write $N_0=F(f,g,x)$. Because $N_0$ is finitely generated, its
characteristic finite-index subgroups are cofinal among its
finite-index subgroups. For any finite target of $Q^0=N_0\rtimes S$,
choose such a characteristic subgroup $U$ inside the restriction
kernel on $N_0$. The action of $S$ on the finite group $N_0/U$
has finite image. Choose a finite-index normal subgroup $V\lhd S$
inside both its action kernel and the restriction kernel of the
given target on $S$. The resulting quotient
\begin{equation}\label{eq:B119-finite-cofinal}
 Q^0\longrightarrow (N_0/U)\rtimes(S/V)
\end{equation}
has kernel contained in the prescribed target kernel. These are
cofinal finite quotients of the whole group. They also show that
$N_0$ has its full induced topology; $S$ does so by the projection
retraction. Taking their inverse limit gives the exact split sequence
\begin{equation}\label{eq:B119-completed-split}
 1\longrightarrow\widehat{N_0}\longrightarrow\widehat{Q^0}
 \longrightarrow\widehat S\longrightarrow1,
 \qquad \widehat{Q^0}=\widehat{N_0}\rtimes\widehat S.
\end{equation}
The characteristic subgroups $U$ are also $\delta$-stable, and
$\delta$ fixes $S$ pointwise, so this comparison respects the
actual cone action.

The ordinary fiber has finite index in
\eqref{eq:B119-orbifold-fiber}; its completion is consequently the
ordinary open subgroup of
\[
 \widehat E=C_2(c)\amalg C_2(\tau)\amalg\widehat{\mathbb Z}(f).
\]
The full-profinite finite-factor centralizer theorem used in B.87,
\citet[Theorem B$'$]{A_HR1985}, gives
$C_{\widehat E}(\tau)=\langle\tau\rangle$. Its ordinary
intersection is trivial. Thus $\widehat{N_0}^{\delta}=1$.
Applying the same unique-decomposition argument to
\eqref{eq:B119-completed-split} gives
$(\widehat{Q^0})^\delta=\widehat S$. Since $Q^0$ is an ordinary
index-two subgroup and $\tau$ splits its extension, this also gives
the completed orbifold equality in
\eqref{eq:B119-orbifold-centralizers}.

Finally, the actual unfilled-to-filled map is
$\mathcal G^0\twoheadrightarrow Q^0$ with kernel $\mathcal N$.
Its full-completion quotient kills the closed normal closure of
those eight meridians, by the universal property on finite targets.
This does not identify $\widehat{\mathcal N}$ with that closure,
does not construct a splitting into $\mathcal G^0$, and does not
lift fixed elements through its kernel. Those questions remain
separate from the exact retained quotient proved here.
\end{proof}
