% A tikzcd body is already tokenized when it occurs inside \diagram or \margin.
% Capture it explicitly and replace top-level alignment ampersands by the PGF
% control-sequence separator. Braced cell contents and ordinary arrays outside
% tikzcd retain their own ampersands. Nothing in the source fragment is changed.
\let\qaOriginalTikzcd\tikzcd
\let\qaOriginalEndTikzcd\endtikzcd
\ExplSyntaxOn
\tl_new:N \l__qa_tikzcd_body_tl
\RenewDocumentEnvironment{tikzcd}{O{} +b}
  {
    \group_begin:
    \tl_set:Nn \l__qa_tikzcd_body_tl {#2}
    \tl_replace_all:Nnn \l__qa_tikzcd_body_tl {&} {\&}
    \qaOriginalTikzcd[ampersand~replacement=\&,#1]
    \tl_use:N \l__qa_tikzcd_body_tl
    \qaOriginalEndTikzcd
    \group_end:
  }
  {}

% Transparent groups are intentional: boxing, \sout, and \text around arbitrary
% arguments would break displays, nested editorial marks, and tikz-cd material.
% Cancellations are shown in red rather than struck through in this QA renderer.
% The source \deleted command remains the authoritative cancellation record.
\NewDocumentCommand{\deleted}{+m}{\begingroup\color{qaDeleted}#1\endgroup}
\NewDocumentCommand{\inserted}{+m}{\begingroup\color{qaInserted}#1\endgroup}
\NewDocumentCommand{\unsure}{+m}{\begingroup\color{qaUnsure}#1\endgroup}
\NewDocumentCommand{\illegible}{m}{%
  \ifmmode
    \text{\color{qaUnsure}[illisible : #1]}%
  \else
    {\color{qaUnsure}[illisible : #1]}%
  \fi}
\NewDocumentCommand{\margin}{+m}{%
  \ifmmode
    \begingroup\color{qaMargin}#1\endgroup
  \else
    \par\begingroup\color{qaMargin}\noindent#1\par\endgroup
  \fi}
\NewDocumentCommand{\diagram}{+m}{\begingroup#1\endgroup}
% article.cls defines \figure as the start of a float. This task instead uses
% a one-argument editorial macro containing an already explicit TikZ picture.
\RenewDocumentCommand{\figure}{+m}{\begingroup#1\endgroup}

\let\qaTextUline\uline
\let\qaTextUuline\uuline
\RenewDocumentCommand{\uline}{+m}{%
  \ifmmode\underline{#1}\else\qaTextUline{#1}\fi}
\RenewDocumentCommand{\uuline}{+m}{%
  \ifmmode\underline{\underline{#1}}\else\qaTextUuline{#1}\fi}
