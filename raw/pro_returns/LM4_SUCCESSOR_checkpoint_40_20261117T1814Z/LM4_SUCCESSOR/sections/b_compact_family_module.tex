\subsection*{The actual compact family module and a finite source-level obstruction}

\statement{B.128}{Proved}{The continuous abelianization of the actual closed family kernel is a free completed group-ring module with paired involution, while the closed source has a proper image in its invariants}
\label{stmt:B128}
Use the actual free presentation and full signed source orbit of B.123:
\[
 F=F(U_1,\ldots,U_8)*T,\qquad T=F(f,g,x),\qquad W=F/N,
\]
where the ordered family generators are
\[
 (U_1,\ldots,U_8)=(a,B,d,e,\bar a,\bar B,\bar d,\bar e),
 \qquad \bar U=cUc.
\]
Let $L=\ker(F\to T)$, $J=L/N$, and let $I\leq J$ be the
actual source-family image generated by all signed source translates
of the four contact words. Bars over subgroups below mean actual
closures in the specified ambient full completion. In particular
\[
 \mathcal J=\ker(\widehat W\to\widehat T),\qquad
 \overline I\leq\mathcal J
\]
are actual closed subgroups, with no identification
$\mathcal J=\widehat J$ assumed. Put
\[
 \Lambda_T=\widehat{\mathbb Z}[[\widehat T]],
 \qquad M=\Lambda_T^{\,8}.
\]
Then the literal retained section and family generators give a
continuous $\widehat T$-module isomorphism
\begin{equation}\label{eq:B128-actual-module}
 \mathcal J^{\mathrm{ab}}_{\mathrm{cont}}\cong M.
\end{equation}
The actual selected involution is
\begin{equation}\label{eq:B128-module-involution}
 \delta(\lambda e_i)=\beta(\lambda)x e_{\bar i},\qquad
 \beta(f)=xgx^{-1},\quad\beta(g)=xfx^{-1},\quad\beta(x)=x^{-1},
\end{equation}
where bar exchanges the two four-label sets and $\beta$ extends
continuously to the completed group ring. As a compact additive
$C_2$-module, $M$ is induced from four coefficient summands.
Its continuous Tate groups vanish in every degree.

Nevertheless the image of $\overline I$ in $M^\delta$ is proper.
This is detected by the finite continuous quotient
\begin{equation}\label{eq:B128-finite-detector}
 M\longrightarrow\bigl(\mathbb F_2[C_2]\bigr)^8
\end{equation}
obtained by reducing coefficients modulo two and the retained
$x$-exponent modulo two. In particular the invariant class
\begin{equation}\label{eq:B128-missing-class}
 z=x e_a+e_{\bar a}
\end{equation}
is outside that source image. This statement concerns an abelian
quotient. It neither supplies a fixed lift of $z$ in $\mathcal J$
nor decides $\mathcal J^\delta=\overline I$.

\begin{proof}
\emph{B.128.1. The actual quotient kernels and the whole relation orbit.}
Every node seed of B.123 is a commutator of two based conjugates
of family peripherals. Each factor belongs to $L$. Hence every
seed lies in $[L,L]$. The group $L$ is normal in $F$, so $[L,L]$
is normal in $F$. Every signed source operator preserves $L$
by the actual filling comparison, and therefore preserves $[L,L]$.
The complete defining normal closure thus satisfies
\begin{equation}\label{eq:B128-all-orbit-derived}
 N\subseteq[L,L].
\end{equation}
This includes every retained conjugate and every signed source
word; no finite orbit is substituted for the defining family.

Let
\[
 \mathcal L=\ker(\widehat F\to\widehat T),\qquad
 \mathcal N=\overline N\subseteq\widehat F,
 \qquad D=\overline{[\mathcal L,\mathcal L]}.
\]
Finite quotients of $F$ killing $L$ are exactly finite quotients
of $T$. The finite-quotient universal property therefore gives
$\mathcal L=\overline L$. The same property and the literal
retained section give
\begin{equation}\label{eq:B128-actual-closed-kernels}
 \widehat W=\widehat F/\mathcal N,
 \qquad \mathcal J=\mathcal L/\mathcal N,
 \qquad \mathcal N\subseteq D.
\end{equation}
The last inclusion follows directly from
\eqref{eq:B128-all-orbit-derived}. The image of $D$ in
$\mathcal J$ is its closed commutator subgroup: commutators map
to commutators, and the image of the compact group $D$ is closed.
Consequently
\begin{equation}\label{eq:B128-abelian-closed-quotient}
 \mathcal J^{\mathrm{ab}}_{\mathrm{cont}}=\mathcal L/D.
\end{equation}
These are identities of actual compact quotients, not an interchange
of completion and abelianization for an arbitrary infinite subgroup.

\emph{B.128.2. The compact-module universal property.}
Define
\[
 \Lambda_T=
 \varprojlim_{n,V}(\mathbb Z/n)[\widehat T/V],
\]
where $V$ runs over open normal subgroups of $\widehat T$ and
$n$ over positive integers ordered by divisibility. We use left
coefficients. If a profinite group acts continuously on a profinite
abelian group $A$, its invariant open subgroups are cofinal.
Indeed, for an open subgroup $B\leq A$, continuity at the compact
set $\widehat T\times\{0\}$ gives an open subgroup $C\leq A$
with $tC\subseteq B$ for every $t\in\widehat T$. Thus
$\bigcap_t tB$ contains $C$ and is an invariant open subgroup.
The finite quotients by these subgroups give the continuous
completed-group-ring action. In particular a choice of eight
vectors of $A$ extends uniquely to a continuous
$\widehat T$-module homomorphism $M\to A$.

There is a continuous homomorphism
\[
 \Phi:\widehat F\longrightarrow M\rtimes\widehat T,
 \qquad U_i\longmapsto(e_i,1),\quad
 f,g,x\longmapsto(0,f),(0,g),(0,x).
\]
Its second projection is the actual retained map. Hence it sends
$\mathcal L$ into the abelian subgroup $M$ and kills $D$, giving
\[
 \overline\Phi:\widehat F/D\longrightarrow M\rtimes\widehat T.
\]
Conversely the retained section splits $\widehat F/D$ as
$(\mathcal L/D)\rtimes\widehat T$. The continuous orbit maps
$t\mapsto t\overline U_i t^{-1}$ define the unique module map
$M\to\mathcal L/D$ taking $e_i$ to $\overline U_i$. Together
with the section it gives
\[
 \Psi:M\rtimes\widehat T\longrightarrow\widehat F/D.
\]
Both composites are the identity on the eight family and three
retained generators. These eleven elements topologically generate
both groups: in $M\rtimes\widehat T$ their conjugates supply all
$t e_i$, whose integral span is dense in $M$. The maps are thus
inverse. Restriction to the kernels of their retained projections,
combined with \eqref{eq:B128-abelian-closed-quotient}, proves
\eqref{eq:B128-actual-module}.

\emph{B.128.3. The paired involution and direct norm solvers.}
The actual free-group involution already satisfies
\[
 \delta(U_i)=xU_{\bar i}x^{-1},\qquad\delta|_T=\beta.
\]
Its continuous extension therefore gives
\eqref{eq:B128-module-involution}. Squaring it is the identity
because $\beta(x)x=1$.
Let $M_+$ be the first four coefficient summands. The map
\[
 M_+\times M_+\longrightarrow M,\qquad
 (u,v)\longmapsto u+\delta(v)
\]
is a continuous isomorphism of additive compact groups under
which $\delta$ exchanges the two factors. The two halves are
closed complementary direct factors. Thus the invariant vectors
are $(u,u)$ and each is a norm. The kernel of the norm consists
of $(u,-u)$, the image of $1-\delta$. These explicit formulas
prove exactness of the two-periodic continuous Tate complex in
both degrees, and hence vanishing in every degree. No nonlinear
fixed-element lifting is implied by this additive calculation.

\emph{B.128.4. A complete signed-operator level invariant.}
For clarity we display the finite marked input that controls every
source word. In the exact Schreier basis
\[
 X_{t,i}=tU_it^{-1}\quad(t\in T),
\]
the positive source operators satisfy
$\alpha_s(U_i)=w_{s,i}U_iw_{s,i}^{-1}$, with retained heads
$h_{s,i}=\pi(w_{s,i})$ as follows:
\begin{equation}\label{eq:B128-head-table}
\begin{array}{c|c|c}
 i&h_{m,i}&h_{v,i}\\\hline
 a&1&f^{-1}\\
 B&f^{-1}g&xg^{-1}x^{-1}\\
 d&1&f^{-1}\\
 e&1&xg^{-2}x^{-1}f\\
 \bar a&f^{-1}g&f^{-1}\\
 \bar B&1&f^{-1}gx^{-1}f^{-1}x\\
 \bar d&f^{-1}g&f^{-1}\\
 \bar e&f^{-1}g&f^{-1}gx^{-1}f^{-2}xg
\end{array}
\end{equation}
The retained positive actions are
\begin{equation}\label{eq:B128-retained-actions}
\begin{array}{c|c|c}
 &\phi_m&\phi_v\\\hline
 f&f^{-1}gf&xgx^{-1}\\
 g&f&f^{-1}gx^{-1}fxg^{-1}f\\
 x&xg^{-1}f&xg^{-1}f
\end{array}
\end{equation}
These are the actual B.123 source operators, including its source
correction in $v$, with their entire nonlinear words retained in
the input. They show that
\[
 \chi_x:T\to\mathbb Z,\qquad
 \chi_x(f)=\chi_x(g)=0,\quad\chi_x(x)=1
\]
is preserved, and that every displayed head has zero $\chi_x$.
The same is true for the two inverse operators: the inverse
peripheral-head identity is
$h_{s^{-1},i}=\phi_s^{-1}(h_{s,i}^{-1})$.
The exact signed word checks give the same positive peripheral
core in every case.

For an arbitrary retained word $t$, write
\[
 \alpha_s(t)w_{s,i}
      =q_{s,t,i}\,\phi_s(t)h_{s,i},\qquad q_{s,t,i}\in L.
\]
The complete nonlinear action is then
\[
 \alpha_s(X_{t,i})=
 q_{s,t,i}X_{\phi_s(t)h_{s,i},i}q_{s,t,i}^{-1}.
\]
Only after abelianizing does the connector disappear. It follows
that every signed source word preserves each color and each
integer level $\chi_x(t)$ of the discrete module. The same
identities extend continuously to the actual compact module and
to its finite level quotients.

\emph{B.128.5. The exact contact vectors and finite obstruction.}
The four full raw contact words of B.123 have the following
Schreier abelianizations, with $e_{t,i}=t e_i$:
\begin{equation}\label{eq:B128-contact-vectors}
\begin{array}{c|c}
 s_1&-e_{1,B}-e_{x,\bar B}\\
 s_2&-e_{f,e}-e_{xg,\bar e}\\
 s_3&-e_{1,d}-e_{x,\bar d}\\
 s_4&-e_{1,a}-e_{x,\bar a}
\end{array}
\end{equation}
This is obtained by scanning each complete raw word from left to
right: a signed family letter contributes the basis element at
its current retained prefix, while a retained letter updates that
prefix. Each row has exactly the two displayed nonzero terms;
there are no suppressed surviving coordinates. The pairing is
literal since $\beta(1)x=x$ and $\beta(f)x=xg$.

The actual subgroup $I$ is generated by all signed source
translates of these four contacts. By B.128.4, its abelian image
is supported on first-half level zero and second-half level one.
To retain this distinction after taking closures, extend
$\chi_x$ to $\widehat T\to\widehat{\mathbb Z}$ and reduce
both that character and the coefficients modulo two. The resulting
map is \eqref{eq:B128-finite-detector}. Every signed source
operator acts trivially on its sixteen color/level coordinates.
The image of $\overline I$ is therefore exactly the span of the
four vectors with first-half level zero and second-half level one,
one for each color. The reverse inclusion follows from the four
actual contact rows themselves, and passage to the closure changes
no image in this finite target.

The class $z$ in \eqref{eq:B128-missing-class} is invariant by
\eqref{eq:B128-module-involution}, but its first-half level is one
and its second-half level is zero. Its finite image has a nonzero
coordinate outside the preceding span. This proves the stated
properness. The argument concerns the image of a subgroup in the
continuous abelianization; it supplies no representative of $z$
in the actual fixed subgroup and no decision of the nonlinear
fixed-source comparison.
\end{proof}

\paragraph{Complete input and independent verification.}
The actual forty node relators, eighty based peripheral factors,
thirty-two signed family images and twelve retained corrections
are recorded in the full family-kernel certificate. The separate
source-level certificate retains every term of the four contact
vectors. Independent evaluators replay all these finite inputs,
including both involution squares, and reconstruct the complete
contact vectors from their literal source words. The all-orbit
inclusion \eqref{eq:B128-all-orbit-derived} and the two inverse
compact maps are the proofs of the infinite and completed claims.
