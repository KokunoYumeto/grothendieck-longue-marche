\subsection*{The actual prefix basis and ordinary regional sections}

\statement{B.115}{Proved}{The prefix automorphism of the actual neck group has no regional extension carrying the base generator to an ordinary element of odd base winding, discretely or in full profinite completion}
\label{stmt:B115}
Let $R$ be the actual annular bad-region group of B.112, with its
specified adapted cone $C$, and write $\bar x=CxC$. Put $B=\bar b$
and $f=hk^{-1}$. The actual common disk group of B.114 is
\[
 K=F(a,B,d,e,f)\leq R.
\]
In its physical meridian basis
\begin{equation}\label{eq:B115-physical-basis}
 \begin{gathered}
 O_1=aBa^{-1},\qquad O_2=a,\qquad
 O_3=fde d^{-1}f^{-1},\\
 O_4=d,\qquad O_5=d^{-1}fd,
 \end{gathered}
\end{equation}
define $P_i=O_1\cdots O_i$ and the free-group automorphism
$\theta:K\to K$ by $\theta(O_i)=P_i$.
Let $\epsilon:R\to\mathbb Z/2$ be the ordinary-cover character,
and let $w:R\to\mathbb Z$ be the geometric annular base winding.
Thus
\begin{equation}\label{eq:B115-characters}
 \begin{array}{c|rrrrrrr}
        &a&b&d&e&C&h&k\\ \hline
 \epsilon&0&0&0&0&1&0&0\\
 w       &0&0&0&0&0&1&1
 \end{array}
 \qquad \epsilon(f)=w(f)=0.
\end{equation}
There is no homomorphism $\Phi:R\to R$ such that
\begin{equation}\label{eq:B115-forbidden-extension}
 \Phi|_K=\theta,\qquad
 \epsilon(\Phi(k))=0,\qquad w(\Phi(k))\equiv1\pmod2.
\end{equation}
The same statement holds for continuous homomorphisms
$\widehat\Phi:\widehat R\to\widehat R$, with the full profinite
completions and continuous extensions of the indicated maps.
In particular this excludes an ordinary regional extension covering
either the identity or inversion on the geometric base generator.

More precisely, there is an explicit epimorphism
\[
 \rho:R\twoheadrightarrow
 T=(\mathbb F_2)^{\{a,b,d,e\}\times\{1,2,3\}}\rtimes S_3,
 \qquad |T|=24576,
\]
whose permutation part $\chi:R\to S_3$ satisfies
\begin{equation}\label{eq:B115-strong-conclusion}
 C_R(\theta(a))\subseteq\ker\chi,
 \qquad
 C_{\widehat R}(\theta(a))\subseteq\ker\widehat\chi.
\end{equation}
Consequently every regional conjugate of $k$, and every correction of
such a conjugate by words in the normal closure of $a,b,d,e$, fails to
centralize $\theta(a)$. The analogous conclusion holds for conjugates
and closed normal-closure corrections in $\widehat R$.

This theorem concerns the unchanged regional group. It does not
preclude rewriting its presentation in the prefix basis while
transforming every relator. It also makes no assertion that $\rho$
extends through the two cross commutators of B.114.

\begin{proof}
\emph{B.115.1. The complete marked regional presentation.}
The reversible actual reduction in B.112 gives generators
$a,b,d,e,C,h,k$, with $f=hk^{-1}$, and the following seventeen
defining relators:
\begin{equation}\label{eq:B115-regional-relations}
 \begin{gathered}
 C^2=1,\qquad [h,C]=1,\qquad hkh^{-1}=CkC,\\
 [k,U]=1\quad(U=a,b,d,e),\\
 [U,CUC]=1\quad(U=a,b,d,e),\\
 [a,b]=[CaC,d]=[CaC,e]=[d,CeC]=1,\\
 [b,hdh^{-1}]=[b,heh^{-1}]=1.
 \end{gathered}
\end{equation}
Here $[x,y]=xyx^{-1}y^{-1}$. The last ten relations are the
simultaneous reduction of all actual filled node relations; the
retained zero loop and all its consequences are already present.
Thus verifying \eqref{eq:B115-regional-relations} defines a map of
the entire regional group, rather than of an isolated nodal subgroup.

The character $\epsilon$ is the actual ordinary covering character:
only the cone is odd among these generators. Both $h$ and $k$ project
to the positive geometric base generator, whereas the six generic
fiber meridians have winding zero. This proves the geometric typing
in \eqref{eq:B115-characters}. In particular $w$ is not the auxiliary
height used to unfold B.112's graph presentation.

For completeness, the inverse of the physical basis change
\eqref{eq:B115-physical-basis} is
\begin{equation}\label{eq:B115-physical-inverse}
 \begin{gathered}
 a=O_2,\qquad B=O_2^{-1}O_1O_2,\qquad d=O_4,\\
 f=O_4O_5O_4^{-1},\qquad
 e=O_5^{-1}O_4^{-1}O_3O_4O_5.
 \end{gathered}
\end{equation}
The prefix substitution has inverse
\[
 \theta^{-1}(O_1)=O_1,\qquad
 \theta^{-1}(O_i)=O_{i-1}^{-1}O_i\quad(2\leq i\leq5).
\]
Both compositions follow by cancellation of consecutive prefix
products. In particular the actual marked word $a=O_2$ satisfies
\begin{equation}\label{eq:B115-theta-a}
 \theta(a)=P_2=(aBa^{-1})a=aB.
\end{equation}
No independent conjugation of the physical meridians is made.

\emph{B.115.2. A quotient of the full regional presentation.}
Let $V$ be the twelve-dimensional vector space over $\mathbb F_2$
with basis $e_{U,j}$, where $U\in\{a,b,d,e\}$ and
$j\in\{1,2,3\}$. The action of $S_3$ is
$p e_{U,j}=e_{U,p(j)}$: it permutes positions and preserves each
named family. Define $T=V\rtimes S_3$ using
\begin{equation}\label{eq:B115-target-law}
 (m,p)(n,q)=(m+p n,pq),\qquad
 (m,p)^{-1}=(-p^{-1}m,p^{-1}),
\end{equation}
where $pq$ means functional composition $p\circ q$.
Writing $s=(23)$ and $t=(12)$, set
\begin{equation}\label{eq:B115-quotient-map}
 \begin{gathered}
 \rho(U)=(e_{U,3},1)\quad(U=a,b,d,e),\\
 \rho(C)=(0,s),\qquad \rho(k)=(0,t),\qquad
 \rho(h)=(0,s),\qquad \rho(f)=(0,st).
 \end{gathered}
\end{equation}
The cone square holds. Since the images of $h$ and $C$ are equal,
$[h,C]=1$ and $hkh^{-1}=CkC$ hold. The permutation $t$ fixes
position $3$, so the four relations $[k,U]=1$ hold. Every entry
in each of the ten remaining commutators is in the abelian normal
subgroup $V$, including the entries conjugated by $C$ or $h$.
This verifies every defining relation in
\eqref{eq:B115-regional-relations} and gives the homomorphism $\rho$.

The images of $C,k$ generate $S_3$. Their conjugates of the four
position-$3$ lamps give all twelve coordinate lamps. Hence $\rho$
is onto, and $|T|=2^{12}\cdot6=24576$.

\emph{B.115.3. The exact finite centralizer.}
Since $B=CbC$, equations \eqref{eq:B115-theta-a} and
\eqref{eq:B115-quotient-map} give
\[
 \rho(\theta(a))=(v,1),\qquad v=e_{a,3}+e_{b,2}.
\]
For arbitrary $m\in V$ and $p\in S_3$, direct multiplication gives
\begin{equation}\label{eq:B115-target-conjugation}
 (m,p)(v,1)(m,p)^{-1}=(pv,1),\qquad
 [(m,p),(v,1)]=(pv-v,1).
\end{equation}
Thus the translation $m$ cannot affect centralization. Because the
two summands of $v$ have distinct named families, $pv=v$ forces
$p(3)=3$ from the $a$-family and $p(2)=2$ from the $b$-family.
These two conditions force $p=1$. Conversely every element of $V$
commutes with $v$. Therefore
\begin{equation}\label{eq:B115-exact-finite-centralizer}
 C_T((v,1))=V\times\{1\}.
\end{equation}
Taking inverse images proves the discrete containment in
\eqref{eq:B115-strong-conclusion}. This excludes all five
nonidentity permutation heads, not only the transpositions.

Let $N=\langle\!\langle a,b,d,e\rangle\!\rangle_R$.
The permutation map $\chi$ kills $N$. More explicitly, setting
$z=Ch$ in the quotient presentation gives
\begin{equation}\label{eq:B115-coarse-map}
 R/N\cong
 (C_2(C)*\mathbb Z(k))\times\mathbb Z(z),\qquad
 \chi(C)=(23),\quad\chi(k)=(12),\quad\chi(z)=1.
\end{equation}
Indeed $[h,C]=1$ becomes $[z,C]=1$, and
$hkh^{-1}=CkC$ becomes $[z,k]=1$; all other relators vanish.
The coarse head of a conjugate of $k$ is a transposition.
Multiplication on either side by an element of $N$ leaves that head
unchanged. This proves the stated correction obstruction without
assuming a normal-form classification of elements of $R$.

\emph{B.115.4. Ordinary sections and the two characters.}
Use additive parity $\operatorname{sgn}:S_3\to\mathbb Z/2$.
The assignments on the seven generating elements give the identity
of homomorphisms
\begin{equation}\label{eq:B115-parity}
 \operatorname{sgn}\chi=\epsilon+(w\bmod2).
\end{equation}
For $a,b,d,e$ both sides are zero; for $C$, $h$, and $k$ both
sides are one. Equivalently $z$ has $(\epsilon,w)=(1,1)$ and
trivial permutation head. Hence any ordinary element $g$ of odd
geometric base winding has odd permutation head. In $S_3$ that head
is a transposition, and \eqref{eq:B115-exact-finite-centralizer}
implies
\begin{equation}\label{eq:B115-odd-winding}
 [g,\theta(a)]\ne1
 \quad\text{if }\epsilon(g)=0\text{ and }w(g)\equiv1\pmod2.
\end{equation}
In $R$ the complete generic presentation has $[k,a]=1$.
If $\Phi|_K=\theta$, its image therefore satisfies
$[\Phi(k),\theta(a)]=1$. This contradicts
\eqref{eq:B115-odd-winding} under the two remaining conditions in
\eqref{eq:B115-forbidden-extension}. In fact the necessary condition
for any such extension is the stronger $\chi(\Phi(k))=1$.
The proposed image of the cone is immaterial to this contradiction.

\emph{B.115.5. Full profinite completion and scope.}
The actual ordinary regional subgroup retracts onto $K$, as proved
by the marked one-family filling and free-factor projection in
B.114. Thus $K$ has its full induced topology in that subgroup.
Since the latter has finite index in $R$, taking the finite-index
core transfers this assertion to $R$. Consequently
$\widehat K\hookrightarrow\widehat R$, and $\theta$ induces the
specified continuous automorphism $\widehat\theta$ of this closed
subgroup.

The finite quotient $\rho$, the character $\epsilon$, and the parity
of $w$ extend continuously to $\widehat R$; $w$ itself extends to
$\widehat w:\widehat R\to\widehat{\mathbb Z}$. The finite
calculation \eqref{eq:B115-exact-finite-centralizer} gives
the completed containment in \eqref{eq:B115-strong-conclusion}.
Identity \eqref{eq:B115-parity} holds on the completion by density.
Applying these facts to the image of $[k,a]=1$ proves the completed
nonextension statement. The argument also applies to arbitrary
completed conjugates and to corrections in the closed normal closure
of $a,b,d,e$, since $\widehat\chi$ kills that closure. This proof
uses a finite quotient, not a presumed completion of an infinite
graph kernel or an additional compact centralizer theorem.

The conclusion does not obstruct a change of generators with all
regional relators transported along it. It obstructs extending the
particular automorphism $\theta$ to the unchanged regional group
with the stated ordinary section condition. Furthermore it is a
statement before B.114's two added cross commutators: neither
extension of $\rho$ to that final quotient nor survival of this
regional obstruction there is asserted. The centralizer of the
actual group after both cross commutators remains a separate question.
\end{proof}

reduction, all twenty-four signed source/old-action comparisons, both

head in \eqref{eq:B115-exact-finite-centralizer}. The initial
transposition-only output is preserved separately from the stronger
complete-head verification. A separate DE implementation verifies the
