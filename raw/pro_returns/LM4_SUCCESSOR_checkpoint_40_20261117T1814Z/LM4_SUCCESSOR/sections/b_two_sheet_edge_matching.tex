\subsection*{The actual two-sheet edge filling and its matching kernels}

\statement{B.134}{Proved}{Both mirror sheets give an exact edge pushout, an ordinary kernel-matching closure, and an independent finite matching-normal criterion}
\label{stmt:B134}
Use the actual ordinary pre-cross star of B.125, with
$j_0=j$, $j_1=CjC$, $K_1=CKC$ and $p=\operatorname{Ad}(j_0)$.
The kernel of mirror parity in that star has the proper presentation
\begin{equation}\label{eq:B134-pre-HNN}
 \begin{gathered}
 V=R^0*_{K}p(R^0),\qquad p(u)=\eta(u)\quad(u\in K),\qquad t=j_0j_1,\\
 \langle V,t\mid tut^{-1}=\sigma(u)\ (u\in K_1)\rangle,
 \qquad \sigma(CuC)=p(C\eta(u)C)\quad(u\in K).
 \end{gathered}
\end{equation}
Let $\theta=\operatorname{Ad}(j_0)$ on $V$; it interchanges the
two vertex groups, and on the pre-cross HNN group sends $t$ to
$t^{-1}$. Put
\begin{equation}\label{eq:B134-actual-words}
 \begin{gathered}
 a_1=CaC,\quad k_1=CkC,\quad f_1=CfC,\\
 v_0=p(aa_1^{-1}),\quad w_0=\theta(v_0)=aa_1^{-1},\\
 s=v_0^{-1}kv_0,\qquad s'=w_0^{-1}p(k)w_0.
 \end{gathered}
\end{equation}
Then the actual final ordinary one-neck group is
\begin{equation}\label{eq:B134-two-cross-presentation}
 \begin{gathered}
 \Gamma_0=\langle V,t\mid
   tut^{-1}=\sigma(u)\ (u\in K_1),\\
   tk_1t^{-1}=s,\quad ts't^{-1}=p(k_1)\rangle.
 \end{gathered}
\end{equation}
Both last equations are required.

Let $E_1=C\langle K,k\rangle C$ be the actual B.117 subgroup,
and set
\[
 (u_1,u_2,u_3,u_4)
   =\bigl(CaC,CdC,CeC,C(f^{-1}Bf)C\bigr).
\]
Its exact presentation is
\begin{equation}\label{eq:B134-E1}
 E_1=\langle f_1,k_1,u_1,u_2,u_3,u_4\mid
      [(f_1k_1)^2,k_1],\ [k_1,u_i]\ (1\leq i\leq4)\rangle.
\end{equation}
Define the five target words in $V$ by
\begin{equation}\label{eq:B134-rhos}
 \rho_0=[(\sigma(f_1)s)^2,s],\qquad
 \rho_i=[s,\sigma(u_i)]\quad(1\leq i\leq4).
\end{equation}
Put
\begin{equation}\label{eq:B134-initial-quotient}
 \begin{gathered}
 N_* =\langle\!\langle\rho_i,\theta(\rho_i)
                         \ (0\leq i\leq4)\rangle\!\rangle_V,
 \qquad V_*=V/N_*,\\
 E=E_1*\langle y\rangle.
 \end{gathered}
\end{equation}
There are genuine homomorphisms $i,j:E\to V_*$ defined on
$E_1$ by its actual inclusion and by $(K_1,k_1)\mapsto
(\sigma(K_1),s)$, respectively, and on the extra generator by
\begin{equation}\label{eq:B134-extra-edge}
 i(y)=s',\qquad j(y)=p(k_1).
\end{equation}
Neither map is presumed injective. With these maps,
\begin{equation}\label{eq:B134-pushout}
 \Gamma_0=\langle V_*,t\mid ti(e)t^{-1}=j(e)\ (e\in E)\rangle.
\end{equation}

Define normal subgroups of $V_*$ by
\begin{equation}\label{eq:B134-saturation}
 \begin{split}
 M_0&=1,\\
 M_{n+1}&=\langle\!\langle M_n,\ i(j^{-1}(M_n)),\
                            j(i^{-1}(M_n))\rangle\!\rangle_{V_*},\\
 M_\infty&=\bigcup_{n\geq0}M_n.
 \end{split}
\end{equation}
Then $V_*/M_\infty$ embeds in \eqref{eq:B134-pushout} as the
base of a proper ordinary HNN extension. In particular
\begin{equation}\label{eq:B134-discrete-base-kernel}
 \ker(V_*\longrightarrow\Gamma_0)=M_\infty.
\end{equation}
No finite stabilization is asserted.

Independently, let $\mathcal B$ consist of the finite-index normal
subgroups $N\triangleleft V_*$ satisfying $i^{-1}(N)=j^{-1}(N)$.
Their associated open normal subgroups of $\widehat V_*$ give
the exact compact criterion
\begin{equation}\label{eq:B134-compact-base-kernel}
 \ker(\widehat V_*\longrightarrow\widehat\Gamma_0)
      =\bigcap_{N\in\mathcal B}\overline N^{\,\widehat V_*}.
\end{equation}
Every such $N$ is the exact restriction of a finite quotient kernel
of $\Gamma_0$. Equality of the right side with
$\overline{M_\infty}^{\,\widehat V_*}$ is not inferred.

Finally the one actual source commutation of B.130, even together
with its mirror, does not suffice for this edge regrouping. If
\begin{equation}\label{eq:B134-known-U}
 U=s_1s_4^{-1}s_1^{-1}
      =a v_0\sigma(a_1)v_0^{-1}\in V,
\end{equation}
then the three words $\rho_2,\rho_3,\rho_4$ remain nontrivial in
\begin{equation}\label{eq:B134-one-source-repair}
 V/\langle\!\langle[k,U],\theta([k,U])\rangle\!\rangle_V.
\end{equation}
They are not identified here with commutators of the corrected
$D_s$ or $E_s$ source coordinates.

\begin{proof}
\emph{B.134.1. Both actual mirror sheets.}
In B.125 the original ordinary star has central vertex $R^0$
and two leaves $Q_0,Q_1$, whose involutions are $j_0,j_1$.
The kernel of their common mirror parity has two $R^0$ vertices
and two edges. Collapsing the $K$ edge gives the actual proper
amalgam $V$ and stable letter $t=j_0j_1$ in
\eqref{eq:B134-pre-HNN}. The remaining edge $K_1$ has precisely
the indicated map $\sigma$; it is the composition of its actual
$C$-marking and the same based $\eta$ at the other vertex.

The single cross relator in the ordinary star is
\[
 r_0=t_0k_1t_0^{-1}k^{-1},\qquad t_0=v_0t.
\]
Passing to the kernel of mirror parity requires the two
transversal conjugates $r_0$ and $j_0r_0j_0$.
This follows directly because the ordinary star is the
semidirect product of its mirror kernel with $\langle j_0\rangle$:
the normal closure restricted to that kernel is generated normally
by these two words. Set
\[
 r_1=tk_1t^{-1}s^{-1},\qquad
 r_2=ts't^{-1}p(k_1)^{-1}.
\]
Literal free multiplication gives
\begin{equation}\label{eq:B134-two-sheet-identities}
 r_0=v_0r_1v_0^{-1},\qquad
 \theta(r_0)=(w_0t^{-1})r_2^{-1}(w_0t^{-1})^{-1}.
\end{equation}
This proves \eqref{eq:B134-two-cross-presentation}, with no
omitted mirror sheet. Conjugation by $j_0$ interchanges the two
normal relator families, so $\theta$ acts on the actual quotient.

The second sheet is not a redundant equation. Kill all neck and
peripheral words and use the two regional base-winding characters.
They give independent generators $k,k'$ from the two vertices,
with $k_1\mapsto k$, $p(k_1)\mapsto k'$ and $v_0,w_0\mapsto1$.
All original edge equations vanish. The two cross equations become
$[t,k]=1$ and $[t,k']=1$. With only the first imposed, the target
is $\mathbb Z^2(t,k)*\mathbb Z(k')$, where the second is a
nontrivial reduced word. Even killing $k$ leaves the nontrivial
commutator in $F(t,k')$. Thus one cannot replace
\eqref{eq:B134-two-cross-presentation} by its first cross equation.

\emph{B.134.2. The actual initial pushout.}
B.117 proves the entire marked presentation
\eqref{eq:B134-E1}, including its retained-square relation.
In \eqref{eq:B134-two-cross-presentation}, conjugating those
five actual relations by $t$ kills exactly the five words
\eqref{eq:B134-rhos}. Applying the actual $\theta$ symmetry
also kills their five mirrors. Hence $N_*$ is killed in
$\Gamma_0$.

The quotient $V_*$ therefore supports the two genuine maps
from $E_1$ described in the statement: its relations hold on the
inclusion side and their target images hold by construction.
The free product with $\langle y\rangle$ extends them by
\eqref{eq:B134-extra-edge}. No presentation of the actual
subgroup $\langle E_1,s'\rangle$ is assumed; the map from
the abstract $E$ may have a kernel.

The $E_1$ part of \eqref{eq:B134-pushout} retains the original
edge equations and the first cross equation. Its $y$ equation
is exactly the second cross equation. Conversely all the
relations used to pass from $V$ to $V_*$ were just proved
consequences of those actual equations. The two presentation
maps fixing $V$ and $t$ are inverse. This proves the exact
pushout \eqref{eq:B134-pushout} before any claim of properness.

\emph{B.134.3. Ordinary kernel matching is exact.}
The sequence in \eqref{eq:B134-saturation} is increasing, so
$M_\infty$ is a normal subgroup of $V_*$. The two induced
maps from $E$ to $V_*/M_\infty$ have the same kernel.
Indeed, if $i(e)\in M_\infty$, it belongs to some $M_n$;
the definition then puts $j(e)$ in $M_{n+1}$. The converse
is identical. Write $Q_E$ for this common kernel.
Consequently both maps
\[
 E/Q_E\longrightarrow V_*/M_\infty
\]
are injective. The ordinary HNN construction on these two
embedded images is proper by its reduced-word normal form.

It is exactly \eqref{eq:B134-pushout}. Inductively every $M_n$
is killed there: if one image of an edge word is already trivial,
the stable-letter equation kills the other image. Thus the
pushout factors through $V_*/M_\infty$. In the opposite
direction the proper HNN extension satisfies all its original
relations. The maps fix the base words and $t$, hence are inverse.
Its base embedding proves \eqref{eq:B134-discrete-base-kernel}.

Equality of the two edge kernels at the initial stage would
already make $M_\infty=1$; injectivity of both initial maps
would be a stronger sufficient condition. Neither follows
from the earlier regional retractions, and neither is asserted.
The entire matching union is retained.

\emph{B.134.4. Every matching finite normal extends exactly.}
Let $\psi:\Gamma_0\to F$ have finite image and let $N$ be the
kernel of its restriction to $V_*$. The stable-letter equation
implies $i(e)\in N$ if and only if $j(e)\in N$. Thus
$N\in\mathcal B$.

Conversely take $N\in\mathcal B$, put $F=V_*/N$ and
$Q_N=i^{-1}(N)=j^{-1}(N)$. The two edge images are isomorphic
copies of the finite group $E/Q_N$ in $F$. Give $F$ its faithful
left regular permutation action on the finite set underlying $F$.
Restricted through either edge map, it is a free action of the
same finite group $E/Q_N$, with the same number of orbits.
Choose a bijection between their orbits and an equivariant
bijection on each paired orbit. The resulting permutation
$t_N$ conjugates the first entire edge action to the second.

Sending the base through its regular action and $t$ to $t_N$
therefore gives a homomorphism
\[
 \Gamma_0\longrightarrow\operatorname{Sym}(F)
\]
whose restriction to $V_*$ has kernel exactly $N$.
This is a concrete finite extension of the prescribed base
quotient, rather than an appeal to residual finiteness of an
unspecified group. It proves both the asserted exact restriction
property and \eqref{eq:B134-compact-base-kernel} by intersecting
all finite quotient kernels.

The family $\mathcal B$ is closed under finite intersections,
since equality of the two preimages is preserved under
intersections. Every matching $N$ contains every $M_n$ by the
same induction as B.134.3. Therefore
\[
 \overline{M_\infty}^{\,\widehat V_*}
       \subseteq\bigcap_{N\in\mathcal B}\overline N^{\,\widehat V_*}.
\]
The reverse inclusion would require an additional finite
compatibility argument for the actual quotient base. It is
not a consequence of the ordinary HNN normal form or of
taking the closure of the countable union.

\emph{B.134.5. The actual one-source repair still misses three leaves.}
Use the actual retraction $r:R^0\to K$, with
\[
 r(CuC)=\zeta(u),\qquad
 \zeta(a)=\zeta(B)=\zeta(d)=\zeta(e)=1,\quad\zeta(f)=f,
 \qquad r(k)=r(CkC)=1.
\]
Let $q:K\to F(a,B,d,e)$ kill $f$. Define $\Phi$ on the
left vertex $R^0$ by its actual base-winding map, killing $K$
and sending $k$ to $z_0\in\mathbb Z(z_0)$. On the right vertex
$p(R^0)$ define it explicitly by
\[
 \Phi(p(g))=q\,r(CgC)\qquad(g\in R^0).
\]
Here $q r\operatorname{Ad}(C)$ kills $K$ and $k$, and sends
$CuC$ to $q(u)$. Both vertex maps kill the common edge $K$,
so they define an actual epimorphism
\begin{equation}\label{eq:B134-Phi}
 \Phi:V\longrightarrow\mathbb Z(z_0)*F(a,B,d,e).
\end{equation}

The certified literal reflection preserves the normal closure
of $f$. Explicitly, with
$P=aBfde d^{-1}f^{-1}d^{-1}$, its word on $f$ is
$\eta(f)=Pf^{-1}P^{-1}$. Since $\eta^2=1$, it induces an
automorphism $\eta_0$ of the four-generator quotient.
Define
\[
 Y_u=a^{-1}\eta_0(u)a\quad(u=a,B,d,e).
\]
They form a free basis: this is the composition of $\eta_0$
with one inner automorphism, with inverse
$\eta_0\operatorname{Inn}(a)$. In particular the literal word
$\eta(a)=aBa^{-1}B^{-1}a^{-1}$ gives
$Y_a=Ba^{-1}B^{-1}$.

The actual images under \eqref{eq:B134-Phi} are
\[
 \Phi(v_0)=a^{-1},\quad\Phi(s)=az_0a^{-1},\quad
 \Phi(\sigma(CuC))=\eta_0(q(u)).
\]
Thus, in their stated order, the four leaf images are
\begin{equation}\label{eq:B134-leaf-detector}
 a[z_0,Y_a]a^{-1},\quad a[z_0,Y_d]a^{-1},\quad
 a[z_0,Y_e]a^{-1},\quad a[z_0,Y_B]a^{-1}.
\end{equation}
The fifth target relation maps to one because
$\Phi(\sigma(f_1))=1$ and $[\Phi(s)^2,\Phi(s)]=1$.
This last observation does not remove that relation from
the actual pushout.

The complete source-word identity in
\eqref{eq:B134-known-U} is the B.130 norm comparison,
using the full raw source marking. It gives $\Phi(U)=Y_a$.
Moreover $\Phi(\theta(k))=\Phi(p(k))=1$, so the mirror of
$[k,U]$ is also killed after imposing only $[z_0,Y_a]=1$.
Hence \eqref{eq:B134-one-source-repair} maps onto
\[
 \mathbb Z^2(z_0,Y_a)*F(Y_d,Y_e,Y_B).
\]
Each of the last three words in \eqref{eq:B134-leaf-detector}
is nontrivial there by ordinary free-product normal form.
This proves the asserted actual repair obstruction. It does
not turn those words into commutators of different source
coordinates without a further marked identity.

\emph{B.134.6. Source-kernel scope.}
The actual pre-cross source $\mathscr F\cong F_6$ of B.133
lies in $V$. Discretely its final kernel is therefore
\[
 \mathscr F\cap\pi_*^{-1}(M_\infty),\qquad
 \pi_*:V\longrightarrow V_*.
\]
For completions, let $\lambda:\widehat{\mathscr F}\to
\widehat V_*$ be the induced map. Its final kernel is exactly
\[
 \lambda^{-1}\!\left(
     \bigcap_{N\in\mathcal B}\overline N^{\,\widehat V_*}
                   \right).
\]
No injection of $\lambda$ is assumed after the initial filling.
B.133 identifies this same actual kernel as the closed
normal-cross intersection in $\widehat A$, where its pre-cross
source embedding has been proved. These different ambient
descriptions are compatible without interchanging a completion,
an intersection, or a limit. The full source presentation and
selected fixed-source equality remain open.
\end{proof}
