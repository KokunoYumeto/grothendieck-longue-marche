\subsection*{A split contact image and its entire normal defect}
\statement{B.170}{Proved}{The actual five-generator contact kernel retracts onto its continuous one-sided image, and the whole kernel of that image is the normal closure of the specified noncommuting contact defect}
\label{stmt:B170}
Use the actual groups and maps of B.169, and abbreviate
\(A=A_s,B=B_s,D=D_s,E=E_s\) inside the source.
Let
\[
 \mathcal K_5=\ker(\pi_+:\mathcal H_5\to\mathcal H_1),\qquad
 r_D=[m,D]=mDm^{-1}D^{-1}.
\]
Then \(\mathcal K_5\le\mathcal I_5\), and the continuous label maps
\begin{equation}\label{eq:B170-section-labels}
 \begin{aligned}
 d_i&\longmapsto A^iDA^{-i},\\
 B_{i,n}&\longmapsto A^im^nBm^{-n}A^{-i},\\
 E_{i,n}&\longmapsto A^im^nEm^{-n}A^{-i}
 \end{aligned}
 \qquad(i,n\in\widehat{\mathbb Z})
\end{equation}
define a continuous section
\[
 \iota:\mathcal U_1\longrightarrow\mathcal I_5,\qquad
 \pi_+\iota=\operatorname{id}_{\mathcal U_1}.
\]
Its closed image \(\mathcal U_{\mathrm{lift}}\) is an actual compact
retract, with
\begin{equation}\label{eq:B170-split-kernel}
 \begin{gathered}
 \mathcal I_5=\mathcal K_5\rtimes\mathcal U_{\mathrm{lift}},\\
 \mathcal K_5=
 \overline{\langle\!\langle[m,D]\rangle\!\rangle}_{\mathcal H_5}
 \ne1.
 \end{gathered}
\end{equation}
In the ordinary category,
\begin{equation}\label{eq:B170-discrete-quotient}
 H_5/\langle\!\langle[m,D]\rangle\!\rangle_{H_5}
       \cong\bigl(F(A,D)\times\mathbb Z_m\bigr)*F(B,E).
\end{equation}
The section is not equivariant for the actual \(J_0\)-conjugation action.
Neither an original presentation of \(H_5\) nor a retraction of all
\(\mathcal I\) or of the ambient group is asserted.

\begin{proof}
\emph{B.170.1. The literal compact section.}
Every value in \eqref{eq:B170-section-labels} lies in \(\mathcal I_5\):
the discarded contacts map to one under the filled one-family map, and
its kernel in \(\mathcal H_5\) is closed and normal. Profinite powers and
multiplication make all three label maps continuous. The relation
\([A,m]=1\) is the actual B.130 relation, and B.154 supplies the marked
full compact \(J_0\)-section; no discontinuous choice of lifts occurs.
The continuous-family universal property proved in B.169 therefore
extends the labels uniquely to \(\iota\).

In the one-sided image, these labels are precisely
\(a^ida^{-i}\), \(a^im^nBm^{-n}a^{-i}\), and
\(a^im^nEm^{-n}a^{-i}\). Hence \(\pi_+\iota\) fixes the entire
topological generating family of \(\mathcal U_1\), and is the identity.
Thus \(\iota\) is injective. Its compact image is closed, has its intrinsic
profinite topology, and is the image of the retraction
\(\iota\pi_+:\mathcal I_5\to\mathcal U_{\mathrm{lift}}\).
This proves the semidirect decomposition in
\eqref{eq:B170-split-kernel}.

\emph{B.170.2. The whole normal kernel.}
Consider the profinite group
\[
 P_0=\widehat{\mathbb Z}^{\,2}_{A,m}
             \amalg\widehat F(B,D,E).
\]
The actual \([A,m]=1\) gives a marked epimorphism
\(P_0\to\mathcal H_5\). Its composite to \(\mathcal H_1\)
has kernel
\[
 R_0=\overline{\langle\!\langle[m,D]\rangle\!\rangle}_{P_0}.
\]
Indeed adding that one relation gives exactly the presentation
\[
 \langle A,D,m,B,E\mid[A,m]=[D,m]=1\rangle_{\mathrm{prof}}
   =(\widehat F(A,D)\times\widehat{\mathbb Z}_m)
          \amalg\widehat F(B,E).
\]
The two presentation maps fix the five marked generators and satisfy
exactly the displayed commuting relations, so they are inverse.
If \(C_0=\ker(P_0\to\mathcal H_5)\), factorization gives
\(C_0\subseteq R_0\). Thus
\[
 \ker(\mathcal H_5\to\mathcal H_1)=R_0/C_0
   =\overline{\langle\!\langle[m,D]\rangle\!\rangle}_{\mathcal H_5}.
\]
This does not assume \(C_0=1\). All unknown original source relations
remain in \(C_0\), inside \(R_0\).

The same argument for
\((\mathbb Z^2_{A,m})*F(B,D,E)\) proves
\eqref{eq:B170-discrete-quotient}: the map to the actual quotient and the
actual one-sided map back are inverse on all marked generators.
The normal closure in \eqref{eq:B170-split-kernel} is in
\(\mathcal H_5\), not merely in \(\mathcal I_5\) without the required
\(J_0\)-translates. B.132 proves that \([m,D]\) is nontrivial both
ordinarily and in the actual ambient completion, so \(\mathcal K_5\ne1\).

\emph{B.170.3. The exact action correction.}
For \(j=A^sm^t\) and \(s,t,i,n\in\widehat{\mathbb Z}\), direct
multiplication gives
\begin{equation}\label{eq:B170-action-defect}
 \begin{aligned}
 j\iota(B_{i,n})j^{-1}&=\iota(B_{i+s,n+t}),\\
 j\iota(E_{i,n})j^{-1}&=\iota(E_{i+s,n+t}),\\
 j\iota(d_i)j^{-1}
   &=\zeta_{i+s,t}\,\iota(d_{i+s}),\\
 \zeta_{h,t}&=A^h[m^t,D]A^{-h}\in\mathcal K_5.
 \end{aligned}
\end{equation}
In particular \(m\) fixes \(d_0\) in the image kernel but does not fix
its chosen lift \(D\). In fact \(mDm^{-1}\) cannot belong to
\(\mathcal U_{\mathrm{lift}}\): its image under \(\pi_+\) is \(d_0\),
and injectivity of \(\pi_+\) on that subgroup would force it to equal \(D\).
The identities retain every actual correction. They give no
\(J_0\)-equivariant section, no internal presentation of
\(\mathcal K_5\), and no full contact-image equality.
\end{proof}
