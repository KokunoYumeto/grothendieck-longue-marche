\[
\boxed{\operatorname{Hom}(F^\bullet,\mathbb{R}f^!(G^\bullet))
\xrightarrow{\sim}

\[
(F^\bullet\in\operatorname{Ob}\mathrm D^b(X),\quad G^\bullet\in\operatorname{Ob}\mathrm D^+(Y)).
